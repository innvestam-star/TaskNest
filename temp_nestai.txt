import React, { useState, useRef, useEffect } from 'react';
import { Send, Sparkles, Zap, CalendarDays, ListTodo, CheckCircle, Clock, Lock } from 'lucide-react';
import { Link } from 'react-router-dom';
import PageHeader from '../components/PageHeader';
import Paywall from '../components/Paywall';
import { breakDownGoal, planMyDay, chat } from '../services/aiService';
import { useSubscription } from '../context/SubscriptionContext';

export default function NestAI() {
    const { canUse, isFree } = useSubscription();
    const [messages, setMessages] = useState([
        {
            id: 1,
            type: 'ai',
            content: "Hi! I'm NestAI, your productivity co-pilot. ðŸš€\n\nI can help you:\nâ€¢ Break down big goals into tasks\nâ€¢ Plan your day optimally\nâ€¢ Suggest the best times for work\n\nTry the quick actions below or just tell me what you need!",
            timestamp: new Date(),
        }
    ]);
    const [input, setInput] = useState('');
    const [isTyping, setIsTyping] = useState(false);
    const [showPaywall, setShowPaywall] = useState(false);
    const [freeTrialMessages, setFreeTrialMessages] = useState(2); // Allow 2 free messages
    const [addedTasks, setAddedTasks] = useState(new Set()); // Track which message tasks have been added
    const messagesEndRef = useRef(null);

    const hasAIAccess = canUse('aiAssistant');

    const scrollToBottom = () => {
        messagesEndRef.current?.scrollIntoView({ behavior: 'smooth' });
    };

    useEffect(() => {
        scrollToBottom();
    }, [messages]);

    const addMessage = (type, content, extra = {}) => {
        setMessages(prev => [...prev, {
            id: Date.now(),
            type,
            content,
            timestamp: new Date(),
            ...extra
        }]);
    };

    const checkAccess = () => {
        if (hasAIAccess) return true;

        if (freeTrialMessages > 0) {
            setFreeTrialMessages(prev => prev - 1);
            return true;
        }

        setShowPaywall(true);
        return false;
    };

    const handleSend = async () => {
        if (!input.trim()) return;

        if (!checkAccess()) return;

        const userMessage = input.trim();
        setInput('');
        addMessage('user', userMessage);
        setIsTyping(true);

        try {
            const lowerInput = userMessage.toLowerCase();

            if (lowerInput.includes('plan my day') || lowerInput.includes('schedule') || lowerInput.includes('plan today')) {
                const result = await planMyDay();
                addMessage('ai', result.message, { schedule: result.schedule });
            } else if (lowerInput.includes('i need') || lowerInput.includes('i want') || lowerInput.includes('help me') || lowerInput.includes('prepare') || lowerInput.includes('create')) {
                const result = await breakDownGoal(userMessage);
                addMessage('ai', result.message, { tasks: result.tasks });
            } else {
                const response = await chat(userMessage);
                addMessage('ai', response);
            }
        } catch (error) {
            addMessage('ai', "I'm sorry, something went wrong. Please try again.");
        } finally {
            setIsTyping(false);
        }
    };

    const handleQuickAction = async (action) => {
        if (!checkAccess()) return;

        addMessage('user', action);
        setIsTyping(true);

        try {
            if (action === 'Plan my day') {
                const result = await planMyDay();
                addMessage('ai', result.message, { schedule: result.schedule });
            } else if (action === 'Break down a goal') {
                addMessage('ai', "Sure! Tell me about your goal. For example:\nâ€¢ 'I need to prepare for my final exams'\nâ€¢ 'I want to launch a new website'\nâ€¢ 'Help me prepare a presentation'");
            } else if (action === 'Show my tasks') {
                addMessage('ai', "You have 5 pending tasks today. Would you like me to help you prioritize them or reschedule any?");
            }
        } finally {
            setIsTyping(false);
        }
    };

    return (
        <div className="flex-1 flex flex-col min-h-0 bg-background transition-colors duration-300">
            <PageHeader
                title="NestAI"
                subtitle="Your productivity co-pilot"
                icon={Sparkles}
                iconClassName="bg-gradient-to-br from-violet-500 to-purple-600 text-white"
            >
                <div className="flex items-center gap-3">
                    {isFree && !hasAIAccess && (
                        <span className="text-xs text-text-muted">
                            {freeTrialMessages > 0 ? `${freeTrialMessages} free messages left` : 'Trial ended'}
                        </span>
                    )}
                    <span className="px-3 py-1 text-xs font-bold bg-gradient-to-r from-violet-500 to-purple-600 text-white rounded-full shadow-lg shadow-purple-500/20">PRO</span>
                </div>
            </PageHeader>

            {/* Free User Banner */}
            {isFree && !hasAIAccess && freeTrialMessages === 0 && (
                <div className="bg-gradient-to-r from-violet-500/10 to-purple-500/10 border-b border-violet-200 px-6 py-3">
                    <div className="flex items-center justify-between max-w-4xl mx-auto">
                        <div className="flex items-center gap-3">
                            <Lock className="w-5 h-5 text-violet-600" />
                            <p className="text-sm text-gray-700">
                                <strong>Upgrade to Pro</strong> to continue using NestAI with unlimited messages
                            </p>
                        </div>
                        <Link to="/pricing" className="px-4 py-1.5 bg-gradient-to-r from-violet-500 to-purple-600 text-white text-sm font-medium rounded-lg hover:shadow-lg transition-all">
                            Upgrade Now
                        </Link>
                    </div>
                </div>
            )}

            {/* Chat Area */}
            <div className="flex-1 overflow-y-auto p-6 space-y-6">
                {messages.map(msg => (
                    <div key={msg.id} className={`flex ${msg.type === 'user' ? 'justify-end' : 'justify-start'}`}>
                        <div className={`max-w-2xl ${msg.type === 'user' ? 'order-2' : ''}`}>
                            {msg.type === 'ai' && (
                                <div className="flex items-center gap-2 mb-2">
                                    <div className="w-6 h-6 rounded-lg bg-gradient-to-br from-violet-500 to-purple-600 flex items-center justify-center text-white text-xs">
                                        <Sparkles className="w-3 h-3" />
                                    </div>
                                    <span className="text-xs font-medium text-gray-500">NestAI</span>
                                </div>
                            )}
                            <div className={`p-4 rounded-2xl whitespace-pre-wrap ${msg.type === 'user'
                                ? 'bg-primary text-white rounded-br-md'
                                : 'bg-white border border-gray-100 shadow-sm text-gray-800 rounded-bl-md'
                                }`}>
                                {msg.content}
                            </div>

                            {/* Render Task Breakdown */}
                            {msg.tasks && (
                                <div className="mt-4 space-y-2">
                                    {msg.tasks.map((task, i) => (
                                        <div key={i} className="flex items-center gap-3 p-3 bg-white border border-gray-100 rounded-xl shadow-sm hover:shadow-md transition-shadow">
                                            <div className="w-5 h-5 rounded-full border-2 border-gray-300"></div>
                                            <div className="flex-1">
                                                <h4 className="text-sm font-medium text-gray-900">{task.title}</h4>
                                                <p className="text-xs text-gray-400">{task.duration}</p>
                                            </div>
                                            <span className={`px-2 py-0.5 text-[10px] font-bold uppercase rounded ${task.priority === 'High' ? 'bg-red-50 text-red-600' :
                                                task.priority === 'Medium' ? 'bg-amber-50 text-amber-600' :
                                                    'bg-blue-50 text-blue-600'
                                                }`}>{task.priority}</span>
                                        </div>
                                    ))}
                                    <button
                                        onClick={() => {
                                            // Add tasks to localStorage
                                            const existingTasks = JSON.parse(localStorage.getItem('tasknest_tasks') || '[]');
                                            const newTasks = msg.tasks.map((task, i) => ({
                                                id: Date.now() + i,
                                                title: task.title,
                                                priority: task.priority,
                                                duration: task.duration,
                                                completed: false,
                                                createdAt: new Date().toISOString(),
                                                source: 'NestAI'
                                            }));
                                            localStorage.setItem('tasknest_tasks', JSON.stringify([...existingTasks, ...newTasks]));
                                            setAddedTasks(prev => new Set([...prev, msg.id]));
                                            alert(`${msg.tasks.length} tasks added to My Tasks!`);
                                        }}
                                        disabled={addedTasks.has(msg.id)}
                                        className="w-full mt-2 py-2.5 bg-primary/10 text-primary font-medium rounded-xl hover:bg-primary/20 transition-colors text-sm cursor-pointer disabled:opacity-50 disabled:cursor-not-allowed"
                                    >
                                        {addedTasks.has(msg.id) ? 'âœ“ Added to Tasks' : 'âœ¨ Add All to My Tasks'}
                                    </button>
                                </div>
                            )}

                            {/* Render Schedule */}
                            {msg.schedule && (
                                <div className="mt-4 bg-white border border-gray-100 rounded-xl shadow-sm overflow-hidden">
                                    {msg.schedule.map((item, i) => (
                                        <div key={i} className={`flex items-center gap-4 p-3 border-b border-gray-50 last:border-0 ${item.type === 'break' ? 'bg-green-50/50' : item.type === 'meeting' ? 'bg-blue-50/50' : ''}`}>
                                            <span className="text-xs font-medium text-gray-400 w-16">{item.time}</span>
                                            <div className={`w-2 h-2 rounded-full ${item.type === 'focus' ? 'bg-purple-500' :
                                                item.type === 'meeting' ? 'bg-blue-500' :
                                                    item.type === 'break' ? 'bg-green-500' :
                                                        'bg-gray-400'
                                                }`}></div>
                                            <span className="text-sm text-gray-800">{item.task}</span>
                                        </div>
                                    ))}
                                </div>
                            )}
                        </div>
                    </div>
                ))}

                {isTyping && (
                    <div className="flex justify-start">
                        <div className="flex items-center gap-2 p-4 bg-white border border-gray-100 rounded-2xl shadow-sm">
                            <div className="flex gap-1">
                                <span className="w-2 h-2 bg-gray-400 rounded-full animate-bounce" style={{ animationDelay: '0ms' }}></span>
                                <span className="w-2 h-2 bg-gray-400 rounded-full animate-bounce" style={{ animationDelay: '150ms' }}></span>
                                <span className="w-2 h-2 bg-gray-400 rounded-full animate-bounce" style={{ animationDelay: '300ms' }}></span>
                            </div>
                            <span className="text-sm text-gray-500">NestAI is thinking...</span>
                        </div>
                    </div>
                )}
                <div ref={messagesEndRef} />
            </div>

            {/* Quick Actions */}
            <div className="px-6 py-3 border-t border-gray-100 bg-white/50 backdrop-blur-sm">
                <div className="flex gap-2 justify-center">
                    {[
                        { label: 'Plan my day', icon: CalendarDays },
                        { label: 'Break down a goal', icon: ListTodo },
                        { label: 'Show my tasks', icon: CheckCircle },
                    ].map(action => (
                        <button
                            key={action.label}
                            onClick={() => handleQuickAction(action.label)}
                            disabled={isFree && freeTrialMessages === 0 && !hasAIAccess}
                            className="flex items-center gap-2 px-4 py-2 bg-gray-100 hover:bg-gray-200 text-gray-700 text-sm font-medium rounded-full transition-colors cursor-pointer disabled:opacity-50 disabled:cursor-not-allowed"
                        >
                            <action.icon className="w-4 h-4" />
                            {action.label}
                        </button>
                    ))}
                </div>
            </div>

            {/* Input Area */}
            <div className="p-4 border-t border-gray-200 bg-white">
                <div className="max-w-4xl mx-auto flex gap-3">
                    <input
                        type="text"
                        placeholder={isFree && freeTrialMessages === 0 && !hasAIAccess ? "Upgrade to Pro to continue..." : "Ask NestAI for help..."}
                        className="flex-1 px-5 py-3 bg-gray-100 border border-transparent rounded-xl focus:bg-white focus:border-primary focus:ring-2 focus:ring-primary/10 outline-none text-gray-800 transition-all disabled:opacity-50"
                        value={input}
                        onChange={(e) => setInput(e.target.value)}
                        onKeyDown={(e) => e.key === 'Enter' && handleSend()}
                        disabled={isFree && freeTrialMessages === 0 && !hasAIAccess}
                    />
                    <button
                        onClick={handleSend}
                        disabled={!input.trim() || isTyping || (isFree && freeTrialMessages === 0 && !hasAIAccess)}
                        className="px-5 py-3 bg-gradient-to-r from-violet-500 to-purple-600 text-white rounded-xl font-medium shadow-lg shadow-purple-500/30 hover:shadow-purple-500/50 transition-all disabled:opacity-50 disabled:cursor-not-allowed cursor-pointer"
                    >
                        <Send className="w-5 h-5" />
                    </button>
                </div>
            </div>

            <Paywall
                isOpen={showPaywall}
                onClose={() => setShowPaywall(false)}
                feature="NestAI assistant"
            />
        </div>
    );
}
